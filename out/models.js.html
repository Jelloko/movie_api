/**
 * @constant mongoose
 * The Mongoose library for MongoDB interaction.
 * @type {Object}
 */
const mongoose = require('mongoose');

/**
 * @constant bcrypt
 * The bcrypt library for hashing and comparing passwords.
 * @type {Object}
 */
const bcrypt = require('bcrypt');

/**
 * @typedef {Object} Genre
 * @property {string} Name - The name of the genre.
 * @property {string} Description - A description of the genre.
 */

/**
 * @typedef {Object} Director
 * @property {string} Name - The name of the director.
 * @property {string} Bio - A short biography of the director.
 */

/**
 * @typedef {Object} Movie
 * @property {string} Title - The title of the movie.
 * @property {string} Description - A description of the movie.
 * @property {Genre} Genre - Details about the movie's genre.
 * @property {Director} Director - Details about the movie's director.
 */
let movieSchema = mongoose.Schema({
    Title: { type: String, required: true },
    Description: { type: String, required: true },
    Genre: {
        Name: String,
        Description: String
    },
    Director: {
        Name: String,
        Bio: String
    }
});

/**
 * @typedef {Object} User
 * @property {string} Name - The user's name.
 * @property {string} Password - The user's hashed password.
 * @property {string} Email - The user's email address.
 * @property {Date} [Birthday] - The user's birthday.
 * @property {Array<ObjectId>} FavoriteMovies - List of the user's favorite movies.
 */
let userSchema = mongoose.Schema({
    Name: { type: String, required: true },
    Password: { type: String, required: true },
    Email: { type: String, required: true },
    Birthday: Date,
    FavoriteMovies: [{ type: mongoose.Schema.Types.ObjectId, ref: 'Movies' }]
});

/**
 * Hashes a given password.
 * @function hashPassword
 * @param {string} password - The password to hash.
 * @returns {string} - The hashed password.
 */
userSchema.statics.hashPassword = (password) => {
    return bcrypt.hashSync(password, 10);
};

/**
 * Validates a given password against the user's stored password hash.
 * @function validatePassword
 * @param {string} password - The password to validate.
 * @returns {boolean} - True if the password is valid, false otherwise.
 */
userSchema.methods.validatePassword = function (password) {
    return bcrypt.compareSync(password, this.Password);
};

/**
 * @constant movies
 * Mongoose model for the "Movies" collection.
 * @type {Model<Movie>}
 */
let movies = mongoose.model('Movies', movieSchema);

/**
 * @constant users
 * Mongoose model for the "Users" collection.
 * @type {Model<User>}
 */
let users = mongoose.model('Users', userSchema);

module.exports.movies = movies;
module.exports.users = users;
