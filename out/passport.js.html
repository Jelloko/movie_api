const passport = require('passport');

const LocalStrategy = require('passport-local').Strategy;

/**
 * @constant Models
 * Contains the Mongoose models for the application.
 * @type {Object}
 */
const Models = require('./models.js');

/**
 * @constant passportJWT
 * Passport library for JWT authentication.
 * @type {Object}
 */
const passportJWT = require('passport-jwt');

/**
 * @constant Users
 * Mongoose model for the "Users" collection.
 * @type {Models<Users>}
 */
let Users = Models.users;

/**
 * @constant JWTStrategy
 * Passport JWT Strategy for handling JWTs.
 * @type {Object}
 */
let JWTStrategy = passportJWT.Strategy;

/**
 * @constant ExtractJWT
 * Extracts JWT from the request headers.
 * @type {Object}
 */
let ExtractJWT = passportJWT.ExtractJwt;

/**
 * Passport Local Strategy for username and password authentication.
 * @function LocalStrategy
 * @param {Object} options - Options for the strategy.
 * @param {string} options.usernameField - The field name for the username.
 * @param {string} options.passwordField - The field name for the password.
 * @param {function} verify - Callback function for verifying the user.
 */
passport.use(
  new LocalStrategy(
    {
      usernameField: 'Name',
      passwordField: 'Password',
    },
    /**
     * Verifies the user by checking the provided username and password.
     * @async
     * @param {string} Name - The username provided by the user.
     * @param {string} password - The password provided by the user.
     * @param {function} callback - The callback function.
     * @returns {Promise<void>} - Resolves when the user is verified or an error occurs.
     */
    async (Name, password, callback) => {
      console.log(`${Name} ${password}`);
      await Users.findOne({ Name: Name })
      .then((user) => {
        if (!user) {
          console.log('incorrect username');
          return callback(null, false, {
            message: 'Incorrect username or password.',
          });
        }
        if (!user.validatePassword(password)) {
          console.log('incorrect password');
          return callback(null, false, { message: 'Incorrect password.' });
        }
        console.log('Done');
        return callback(null, user);
      })
      .catch((error) => {
        if (error) {
          console.log(error);
          return callback(error);
        }
      });
    }
  )
);

passport.use(new JWTStrategy({
  jwtFromRequest: ExtractJWT.fromAuthHeaderAsBearerToken(),
  secretOrKey: 'your_jwt_secret'

/**
 * Verifies the user based on the provided JWT payload.
 * @async
 * @param {Object} jwtPayload - The decoded JWT payload.
 * @param {function} callback - The callback function.
 * @returns {Promise<void>} - Resolves when the user is verified or an error occurs.
 */
async (jwtPayload, callback) => {
  return await Users.findById(jwtPayload._id)
    .then((user) => {
      return callback(null, user);
    })
    .catch((error) => {
      return callback(error);
    });
}));
